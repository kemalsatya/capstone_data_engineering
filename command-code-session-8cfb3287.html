"""
Data Ingestion Loaders for Capstone Data Engineering
Streams transformed/cleaned data from workspace/data/Sumber Data/ into PostgreSQL raw.* tables
using high-performance PostgreSQL COPY protocol.
"""

import os

import subprocess
import xml.etree.ElementTree as ET

BASE_DIR = Path(__file__).resolve().parents[2] / "data" / "Sumber Data"
PROJECT_ROOT = Path(__file__).resolve().parents[2]

NULL_LITERALS = {"", "-", "--", "...", "....", "na", "n/a", "null", "none", "nan", "#n/a", "#value!", "#ref!"}

def is_null_val(v):
    if v is None:
        return True
    s = str(v).strip()
    return s.lower() in NULL_LITERALS

def get_psql_cmd():
    """Returns the base command to pipe COPY data into PostgreSQL."""
    return ["docker", "compose", "exec", "-T", "postgres", "psql", "-U", "labuser", "-d", "labdata"]

def copy_from_stream(table_name, columns, row_generator):
    r"""
    Executes PostgreSQL \copy from STDIN streaming rows generated by row_generator.
    """
    col_str = ", ".join(columns)
    copy_sql = f"\\copy {table_name} ({col_str}) FROM STDIN WITH (FORMAT CSV, HEADER FALSE, NULL '\\N');"
    
    cmd = get_psql_cmd() + ["-c", copy_sql]
    
    # Run subprocess streaming stdin
    proc = subprocess.Popen(
        cmd,
        cwd=str(PROJECT_ROOT),
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8"
    )
    
    writer = csv.writer(proc.stdin, lineterminator="\n")
    row_count = 0
    try:
        for row in row_generator():
            # Format row: replace empty string, missing symbols, or None with \N for NULL
            cleaned_row = ["\\N" if is_null_val(v) else str(v).strip() for v in row]
            writer.writerow(cleaned_row)
            row_count += 1
            if row_count % 50000 == 0:
                print(f"      ... streamed {row_count:,} rows")
        proc.stdin.close()
        stdout, stderr = proc.communicate()
        if proc.returncode != 0:
            raise RuntimeError(f"psql COPY failed for {table_name}: {stderr}")
        print(f"    [OK] Loaded {row_count:,} rows into {table_name}")
    except Exception as e:
        proc.kill()
        raise e
    return row_count

        if 'xl/sharedStrings.xml' in zf.namelist():

            sst = [t.text for t in ET.fromstring(sst_xml).iter('{http://schemas.openxmlformats.org/spreadsheetml/2006/main}t')]

        # Skip header row 0
        for r in rows_elem[1:]:
            cells = []
            for c in r.findall('{http://schemas.openxmlformats.org/spreadsheetml/2006/main}c'):
                t = c.attrib.get('t')
                v = c.find('{http://schemas.openxmlformats.org/spreadsheetml/2006/main}v')
                val = v.text if v is not None else ""
                if t == 's' and val != "" and int(val) < len(sst):

                cells.append(val)
            # pad if missing
            while len(cells) < 12:
                cells.append("")
            yield cells[:12] + [filename]

    print(f"--> Ingesting SRC-02: {filename}...")
    return copy_from_stream(table, columns, gen)

    ]

        year_str = "".join([c for c in fpath.stem if c.isdigit()])[-4:]
        year_val = int(year_str) if year_str else None

        def gen():
            with open(fpath, "r", encoding="utf-8-sig", errors="replace") as f:
                reader = csv.reader(f)
                next(reader, None) # skip header
                for row in reader:
                    if len(row) >= 6 and row[0].strip():
                        # skip footer/summary lines if any
                        prov = row[0].strip()
                        if "indonesia" in prov.lower() or "catatan" in prov.lower():
                            continue
                        yield [prov, row[1].strip(), row[2].strip(), row[3].strip(), row[4].strip(), row[5].strip(), year_val, fpath.name]

        print(f"--> Ingesting SRC-12: {fpath.name} (Tahun {year_val})...")
        total += copy_from_stream(table, cols, gen)
    return total

def truncate_tables(sources):
    tables = []
    for s in sources:
        tables.extend(TABLE_MAP.get(s, []))
    if not tables:
        return
    print(f"--> Truncating tables before load: {', '.join(tables)}...")
    import subprocess
    cmd = loaders.get_psql_cmd() + ["-c", f"TRUNCATE {', '.join(tables)} RESTART IDENTITY;"]
    res = subprocess.run(cmd, cwd=str(loaders.PROJECT_ROOT), capture_output=True, text=True)
    if res.returncode != 0:
        print(f"Warning: Truncate failed: {res.stderr}")

        print("    [OK] Tables truncated.")
